const fs = require('node:fs');
const path = require('node:path');
const { randomUUID } = require('node:crypto');
const channel = __dirname;
const lock = path.join(channel, 'lock');
const id = randomUUID();
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
async function main() {
  fs.mkdirSync(lock);
  try {
    fs.writeFileSync(path.join(channel, 'request.tmp'), JSON.stringify({ id }));
    fs.renameSync(path.join(channel, 'request.tmp'), path.join(channel, 'request.json'));
    const deadline = Date.now() + 360000;
    while (Date.now() < deadline) {
      try {
        const response = JSON.parse(fs.readFileSync(path.join(channel, 'response.json'), 'utf8'));
        if (response.id === id) {
          process.stdout.write(response.output + '\n');
          process.exitCode = response.passed ? 0 : 1;
          return;
        }
      } catch (error) {
        if (error.code !== 'ENOENT') throw error;
      }
      await wait(100);
    }
    throw new Error('Repair runner did not respond before the time limit.');
  } finally {
    fs.rmdirSync(lock);
  }
}
main().catch((error) => { console.error(error.message); process.exitCode = 1; });
